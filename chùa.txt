    <script>
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');

        // Danh sách Phật tử và Nhà sư di chuyển dưới chánh điện
        let characters = [
            { x: 250, y: 1600, speed: 0.6, type: 'phattu', direction: 1, text: 'Nam Mô A Di Đà Phật' },
            { x: 800, y: 1700, speed: 0.4, type: 'su', direction: -1, text: 'Thiện tai, thiện tai...' },
            { x: 500, y: 1550, speed: 0.5, type: 'phattu', direction: 1, text: 'Cầu bình an gia đạo' }
        ];

        // Hiệu ứng cánh hoa sen rơi từ trên xuống
        let lotusPetals = [];
        for(let i=0; i<30; i++) {
            lotusPetals.push({
                x: Math.random() * 1080,
                y: Math.random() * 1920,
                size: Math.random() * 15 + 10,
                speedY: Math.random() * 1.5 + 0.8,
                speedX: Math.sin(i) * 0.8,
                rotation: Math.random() * 360
            });
        }

        function drawScene() {
            let gradient = ctx.createLinearGradient(0, 0, 0, 1920);
            gradient.addColorStop(0, '#1c1008');
            gradient.addColorStop(0.5, '#3d2314');
            gradient.addColorStop(1, '#150b05');
            ctx.fillStyle = gradient;
            ctx.fillRect(0, 0, 1080, 1920);

            ctx.fillStyle = 'rgba(255, 215, 0, 0.18)';
            ctx.beginPath();
            ctx.arc(540, 450, 380, 0, Math.PI * 2);
            ctx.fill();

            ctx.fillStyle = 'rgba(255, 255, 255, 0.05)';
            ctx.beginPath();
            ctx.arc(540 + Math.sin(Date.now() / 1000) * 50, 700 - (Date.now() / 25) % 500, 75, 0, Math.PI * 2);
            ctx.fill();

            ctx.fillStyle = 'rgba(255, 182, 193, 0.8)';
            lotusPetals.forEach(p => {
                ctx.save();
                ctx.translate(p.x, p.y);
                ctx.rotate(p.rotation);
                ctx.beginPath();
                ctx.ellipse(0, 0, p.size, p.size / 2, 0, 0, Math.PI * 2);
                ctx.fill();
                ctx.restore();

                p.y += p.speedY;
                p.x += p.speedX;
                p.rotation += 0.01;

                if (p.y > 1920) { p.y = -50; p.x = Math.random() * 1080; }
            });

            characters.forEach(char => {
                ctx.save();
                ctx.translate(char.x, char.y);

                ctx.fillStyle = (char.type === 'su') ? '#ff8c00' : '#8b0000';
                ctx.fillRect(-25, -70, 50, 100);

                ctx.fillStyle = '#ffdbac';
                ctx.beginPath();
                ctx.arc(0, -90, 20, 0, Math.PI * 2);
                ctx.fill();

                ctx.fillStyle = 'rgba(0, 0, 0, 0.75)';
                ctx.fillRect(-100, -150, 200, 45);
                ctx.strokeStyle = '#d4af37';
                ctx.strokeRect(-100, -150, 200, 45);

                ctx.fillStyle = '#fff';
                ctx.font = '16px Arial';
                ctx.textAlign = 'center';
                ctx.fillText(char.text, 0, -122);

                ctx.restore();

                char.x += char.speed * char.direction;
                if (char.x > 900 || char.x < 180) {
                    char.direction *= -1;
                }
            });

            requestAnimationFrame(drawScene);
        }
        drawScene();

        // Kho dữ liệu quẻ xăm đa dạng
        const khoDuLieuXam = {
            cauan: [
                { type: "🌟 CẦU AN BÌNH AN (Thượng Cát)", text: "Gia đạo thuận hòa, tâm thanh tịnh, tai qua nạn khỏi, phước lộc tự khắc tề tựu." },
                { type: "🌟 CẦU AN GIA ĐẠO (Thượng Thượng)", text: "Trên dưới một lòng, nhà cửa êm ấm, mùa màng bội thu, vạn sự bình an vô sự." }
            ],
            tinhduyen: [
                { type: "💖 TÌNH DUYÊN (Thượng Thượng)", text: "Hoa nở mùa xuân, đôi lứa tri kỷ, duyên lành đưa lối, sớm ngày chung đôi vẹn tròn." },
                { type: "⚠️ TÌNH DUYÊN (Hạ Hạ)", text: "Cẩn trọng lời ăn tiếng nói trong tình cảm, tránh ghen tuông vô cớ mà lỡ dở tơ duyên." }
            ],
            sunghiep: [
                { type: "📈 SỰ NGHIỆP / CÔNG DANH (Thượng Thượng)", text: "Cá chép hoá rồng, thăng quan tiến chức, gặp quý nhân phù trợ, mưu sự đại thành." },
                { type: "📉 SỰ NGHIỆP (Hạ Hạ - Nhắc Nhở)", text: "Đường công danh trắc trở tạm thời, chớ nên đầu tư mạo hiểm, thủ thế an toàn là thượng sách." }
            ],
            khaitruong: [
                { type: "🚀 KHAI TRƯƠNG / KINH DOANH (Thượng Thượng)", text: "Buôn may bán đắt, tiền vô như nước sông Tiền, khách vào nườm nượp, phát tài phát lộc!" }
            ]
        };

        // Thêm bình luận vào khung Chat Ticker
        function hienThiBinhLuanChat(username, message) {
            const chatStream = document.getElementById('chat-stream');
            const bubble = document.createElement('div');
            bubble.className = 'chat-bubble';
            bubble.innerHTML = `<span class="chat-user">${username}:</span> ${message}`;
            chatStream.appendChild(bubble);

            if (chatStream.children.length > 6) {
                chatStream.removeChild(chatStream.firstChild);
            }
        }

        // Hàng đợi xử lý tương tác & Quà tặng
        let waitingQueue = [];
        let isProcessing = false;

        function nhanTuongTac(username, keyword, isGift, giftName = "") {
            hienThiBinhLuanChat(username, giftName ? `Đã cúng dường [${giftName}] 🌸` : keyword);

            let data = { name: username, kw: keyword.toLowerCase().trim(), isGift: isGift, gift: giftName };
            if (isGift) {
                waitingQueue.unshift(data);
            } else {
                waitingQueue.push(data);
            }
            if (!isProcessing) xuLyHangDoi();
        }

        function xuLyHangDoi() {
            if (waitingQueue.length === 0) {
                isProcessing = false;
                return;
            }
            isProcessing = true;
            let current = waitingQueue.shift();

            let danhSachChon = khoDuLieuXam.cauan;
            let tieuDeBox = "✨ THỈNH XĂM CẦU AN ✨";

            if (current.isGift) {
                tieuDeBox = `🪷 CẢM TẠ TỊNH TÀI (${current.gift}) 🪷`;
            } else if (current.kw.includes('tinhduyen') || current.kw.includes('duyen')) {
                danhSachChon = khoDuLieuXam.tinhduyen;
                tieuDeBox = "💖 QUẺ XĂM TÌNH DUYÊN 💖";
            } else if (current.kw.includes('sunghiep') || current.kw.includes('congdanh')) {
                danhSachChon = khoDuLieuXam.sunghiep;
                tieuDeBox = "📈 QUẺ XĂM CÔNG DANH 📈";
            } else if (current.kw.includes('khaitruong') || current.kw.includes('buonmay')) {
                danhSachChon = khoDuLieuXam.khaitruong;
                tieuDeBox = "🚀 GIEO QUẺ KHAI TRƯƠNG 🚀";
            }

            let quChon = danhSachChon[Math.floor(Math.random() * danhSachChon.length)];

            const box = document.getElementById('xam-box');
            document.getElementById('box-title').innerText = tieuDeBox;
            document.getElementById('lbl-user').innerText = current.name;
            document.getElementById('lbl-type').innerText = current.isGift ? "🌟 ƯU TIÊN PHƯỚC LỘC VIÊN MÃN 🌟" : quChon.type;
            document.getElementById('lbl-qu').innerText = current.isGift ? "Đạo hữu thành tâm cúng dường tịnh tài, tâm nguyện sở cầu như ý phát tài!" : quChon.text;
            box.style.opacity = '1';

            let loiThoai = current.isGift 
                ? `Nam mô A Di Đà Phật. Thầy vô cùng cảm tạ đạo hữu ${current.name} đã cúng dường ${current.gift}. Nguyện cầu gia đạo bình an, tài lộc hưng vượng!`
                : `Nam mô A Di Đà Phật. Thầy xin hồi đáp cho con ${current.name}. Quẻ của con là ${quChon.type}. ${quChon.text}`;
            
            docGiongMienNam(loiThoai);

            setTimeout(() => {
                box.style.opacity = '0';
                setTimeout(xuLyHangDoi, 1000);
            }, 8000);
        }

        function docGiongMienNam(text) {
            if ('speechSynthesis' in window) {
                window.speechSynthesis.cancel();
                let utterance = new SpeechSynthesisUtterance(text);
                utterance.lang = 'vi-VN';
                utterance.pitch = 0.95;
                utterance.rate = 1.0;
                window.speechSynthesis.speak(utterance);
            }
        }

        // ==========================================
        // KẾT NỐI WEBSOCKET NHẬN CHAT THẬT TỪ TIKTOK LIVE
        // ==========================================
        try {
            // Kết nối qua tool bắt chat trung gian (Ví dụ: TikTok Live Webcast Port mặc định 8888)
            let ws = new WebSocket('ws://localhost:8888');
            
            ws.onmessage = function(event) {
                let packet = JSON.parse(event.data);
                
                // Khi có khán giả bình luận (Chat event)
                if (packet.type === 'chat') {
                    let user = packet.uniqueId;
                    let text = packet.comment;
                    
                    // Kiểm tra nếu comment có chứa từ khóa xin xăm thì đưa vào hệ thống
                    if (text.includes('cauan') || text.includes('tinhduyen') || text.includes('sunghiep') || text.includes('khaitruong') || text.includes('binhan')) {
                        nhanTuongTac(user, text, false);
                    } else {
                        // Chỉ hiện bình luận lên khung chat trôi nổi thông thường
                        hienThiBinhLuanChat(user, text);
                    }
                }
                
                // Khi có khán giả tặng quà (Gift event)
                if (packet.type === 'gift' && packet.repeatEnd) {
                    let user = packet.uniqueId;
                    let giftName = packet.giftName;
                    nhanTuongTac(user, "tangqua", true, giftName);
                }
            };
            
            ws.onopen = function() {
                console.log("Đã kết nối thành công với cổng dữ liệu TikTok Live!");
            };
        } catch (e) {
            console.log("Chưa bật tool kết nối WebSocket, chạy chế độ giả lập.");
            // [TEST GIẢ LẬP KHI KHÔNG CHẠY TOOL]
            setTimeout(() => { nhanTuongTac("Minh Tâm", "cauan", false); }, 2000);
        }
